		skip_build_list

		# Validation
		validate_conflicts_war_goals_all

	execution_priority = 60

	possible = {

		}
		scope:target_country = {
			OR = {
				has_modifier = gw_central_powers_leader

				has_modifier = gw_great_war_leader
				has_variable = gw_war_side_1_leader

		is_significant_demand = yes
	}
