# Stalemate / status quo
gw_war.1006 = {
	type = country_event

	title = gw_war.1006.t
	desc = gw_war.1006.d
	flavor = gw_war.1006.f

	event_image = {
		video = "gfx/event_pictures/europenorthamerica_capitalists_meeting.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	trigger = {
	}

	immediate = {
		every_country = {
			limit = {
				has_modifier = gw_great_war_leader
			}
			remove_modifier = gw_great_war_leader
		}
		gw_clear_war_state = yes
		# Global changes — only first country to process (guard: gw_great_war_active still exists)
		if = {
			limit = { has_global_variable = gw_great_war_active }
			remove_global_variable = gw_great_war_active
			set_global_variable = gw_great_war_concluded
			if = { limit = { has_global_variable = gw_system_active } remove_global_variable = gw_system_active }
			if = {
				limit = { has_global_variable = gw_war_outcome_determined }
				remove_global_variable = gw_war_outcome_determined
			}
			# Clean up Call to War tracking
			if = {
				limit = { has_global_variable = gw_call_to_war_active }
				remove_global_variable = gw_call_to_war_active
			}
			if = {
				limit = { has_global_variable = gw_call_to_war_months }
				remove_global_variable = gw_call_to_war_months
			}
			set_global_variable = { name = gw_world_tension value = 0 }
			# Once the Great War has ended, pre-war tension and crisis bonuses no longer apply.
			gw_clear_all_tension_event_modifiers = yes
		}
		trigger_event = { id = gw_hist.3004 days = 2 }
		# Per-country cleanup (safe to run multiple times)
		gw_clear_crisis_flags = yes
	}

	option = {
		name = gw_war.1006.a
		default_option = yes
	}
}

# --- Hidden: Loser cleanup ---
gw_war.2001 = {
	type = country_event
	hidden = yes

	immediate = {
		# Subjects become independent
		every_country = {
			limit = {
				is_subject_of = root
			}
			make_independent = yes
		}
		# Install a friendly regime aligned with the victorious faction leader.
		gw_apply_regime_change_from_great_war_winner = yes
		# Apply defeat modifier
		if = {
			limit = {
				NOT = { has_modifier = gw_major_defeat }
				NOR = {
					has_variable = gw_war_side_2_leader
					has_variable = gw_war_side_1_leader
				}
				country_rank >= rank_value:great_power
			}
			add_modifier = {
				name = gw_major_defeat
				years = 10
			}
		}
		if = {
			limit = {
				NOT = { has_modifier = gw_major_leader_defeat }
				OR = {
					has_variable = gw_war_side_2_leader
					has_variable = gw_war_side_1_leader
				}
			}
			add_modifier = {
				name = gw_major_leader_defeat
				years = 10
			}
			# Trigger "Stabbed in the Back" myth for the losing leader
			trigger_event = { id = gw_war.1007 days = 30 }
		}
		# War Reparations — losers pay
		if = {
			limit = { country_rank >= rank_value:great_power }
			add_modifier = {
				name = gw_war_reparations_paying
				years = 10
			}
		}
		# Remove order status
		if = {
			limit = { has_modifier = gw_dominating_world_order }
			remove_modifier = gw_dominating_world_order
		}
		if = {
			limit = { has_modifier = gw_great_war_leader }
			remove_modifier = gw_great_war_leader
		}
		gw_clear_war_state = yes
		# NOTE: Do NOT remove gw_great_war_active here — only the winner event (2002)
		# handles global cleanup. If 2001 removes it first, 2002's global section
		# (tension reset, league rotation) would never run.
		gw_clear_crisis_flags = yes
		trigger_event = { id = gw_war.1002 days = 1 }
		trigger_event = { id = gw_hist.3002 days = 2 }
	}
}

# --- Hidden: Winner cleanup ---
gw_war.2002 = {
	type = country_event
	hidden = yes

	immediate = {
		# Apply victory modifier
		if = {
			limit = {
				NOT = { has_modifier = gw_major_victory }
				NOR = {
					has_variable = gw_war_side_2_leader
					has_variable = gw_war_side_1_leader
				}
				country_rank >= rank_value:great_power
			}
			add_modifier = {
				name = gw_major_victory
				years = 10
			}
		}
		if = {
			limit = {
				NOT = { has_modifier = gw_major_leader_victory }
				OR = {
					has_variable = gw_war_side_2_leader
					has_variable = gw_war_side_1_leader
				}
			}
			add_modifier = {
				name = gw_major_leader_victory
				years = 10
			}
		}
		# War Reparations — winners receive
		if = {
			limit = { country_rank >= rank_value:great_power }
			add_modifier = {
				name = gw_war_reparations_receiving
				years = 10
			}
		}
		# Winners dominate the new order
		if = {
			limit = {
				NOT = { has_modifier = gw_dominating_world_order }
				country_rank >= rank_value:great_power
			}
			add_modifier = gw_dominating_world_order
		}
		if = {
			limit = { has_modifier = gw_great_war_leader }
			remove_modifier = gw_great_war_leader
		}
		# Global changes — only first winner to process.
		# Also runs for diplomatic-play backdowns, where gw_great_war_active may never be set.
		if = {
			limit = {
				OR = {
					has_global_variable = gw_great_war_active
					has_global_variable = gw_war_outcome_determined
				}
			}
			if = {
				limit = {
					has_global_variable = gw_great_war_active
					NOT = { has_global_variable = gw_hist_neutral_epilogue_sent }
				}
				set_global_variable = gw_hist_neutral_epilogue_sent
				every_country = {
					limit = {
						gw_is_major_or_greater = yes
						NOT = { has_variable = gw_great_war_participant }
						NOT = { has_variable = gw_war_side_1 }
						NOT = { has_variable = gw_war_side_2 }
						NOT = { has_variable = gw_war_side_1_leader }
						NOT = { has_variable = gw_war_side_2_leader }
						NOT = { has_variable = gw_hist_postwar_seen }
					}
					trigger_event = { id = gw_hist.3003 days = 2 }
				}
			}
			if = {
				limit = { has_global_variable = gw_great_war_active }
				remove_global_variable = gw_great_war_active
			}
			set_global_variable = gw_great_war_concluded
			if = { limit = { has_global_variable = gw_system_active } remove_global_variable = gw_system_active }
			if = {
				limit = { has_global_variable = gw_war_outcome_determined }
				remove_global_variable = gw_war_outcome_determined
			}
			# Clean up Call to War tracking
			if = {
				limit = { has_global_variable = gw_call_to_war_active }
				remove_global_variable = gw_call_to_war_active
			}
			if = {
				limit = { has_global_variable = gw_call_to_war_months }
				remove_global_variable = gw_call_to_war_months
			}
			set_global_variable = { name = gw_world_tension value = 0 }
			# Once the Great War has ended, pre-war tension and crisis bonuses no longer apply.
			gw_clear_all_tension_event_modifiers = yes
		}
		gw_clear_war_state = yes
		gw_clear_crisis_flags = yes
		trigger_event = { id = gw_war.1001 days = 1 }
		trigger_event = { id = gw_hist.3001 days = 2 }
	}
}
