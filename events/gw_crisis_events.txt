# Bloc leaders choose whether to embrace or restrain the arms race.
gw_crisis.1001 = {
	type = country_event

	title = gw_crisis.1001.t
	desc = gw_crisis.1001.d
	flavor = gw_crisis.1001.f

	event_image = {
		video = "gfx/event_pictures/unspecific_military_parade.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		country_rank >= rank_value:major_power
		gw_is_great_war_bloc_leader = yes
		NOT = { has_variable = gw_arms_race_seen }
		global_var:gw_world_tension >= 60
		NOT = { has_global_variable = gw_great_war_active }
	}

	immediate = {
		set_variable = gw_arms_race_seen
	}

	option = {
		name = gw_crisis.1001.a
		default_option = yes
		custom_tooltip = { text = gw_crisis_1001_a_tt }
		add_modifier = {
			name = gw_arms_race
			months = 24
		}
		add_modifier = {
			name = gw_general_staff_program
			months = 18
		}
		trigger_event = { id = gw_crisis.1002 months = 6 }
		# Launch the four-event Arms Race atmosphere chain on this bloc leader.
		trigger_event = { id = gw_crisis.1100 months = 6 }
		ai_chance = {
			base = 10
			modifier = {
				trigger = { has_variable = gw_dissatisfied_with_order }
				add = 12
			}
			modifier = {
				trigger = { global_var:gw_world_tension >= 75 }
				add = 8
			}
		}
	}

	option = {
		name = gw_crisis.1001.b
		custom_tooltip = { text = gw_crisis_1001_b_tt }
		add_modifier = {
			name = gw_arms_limitation_talks
			months = 12
		}
		change_global_variable = { name = gw_world_tension add = -1 }
		ai_chance = {
			base = 8
			modifier = {
				trigger = { has_variable = gw_satisfied_with_order }
				add = 10
			}
			modifier = {
				trigger = { global_var:gw_world_tension >= 80 }
				add = -6
			}
		}
	}
}

# ============================================================
# ARMS RACE ATMOSPHERE CHAIN (Phase 1.5, embrace path only)
# Launched by gw_crisis.1001 embrace option. Four milestone events
# at +6/+12/+18/+24 months. Hard guard: chain stops if arms_race
# modifier expires/removed, tension drops below 60, or Great War starts.
# Each event also echoes to non-leader Major Powers (gw_crisis.111x).
# ============================================================

# Milestone 1 — Naval Estimates (+6 months from embrace)
gw_crisis.1100 = {
	type = country_event

	title = gw_crisis.1100.t
	desc = gw_crisis.1100.d
	flavor = gw_crisis.1100.f

	event_image = {
		video = "gfx/event_pictures/unspecific_military_parade.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		country_rank >= rank_value:major_power
		gw_is_great_war_bloc_leader = yes
		has_modifier = gw_arms_race
		global_var:gw_world_tension >= 60
		NOT = { has_global_variable = gw_great_war_active }
	}

	option = {
		name = gw_crisis.1100.a
		default_option = yes
		add_modifier = {
			name = gw_naval_estimates
			months = 18
		}
		# Echo to non-leader Major Powers
		every_country = {
			limit = {
				country_rank >= rank_value:major_power
				NOT = { gw_is_great_war_bloc_leader = yes }
				NOT = { has_global_variable = gw_great_war_active }
				global_var:gw_world_tension >= 60
			}
			trigger_event = { id = gw_crisis.1110 days = 7 }
		}
		# Chain next event in 6 months
		trigger_event = { id = gw_crisis.1101 months = 6 }
		ai_chance = {
			base = 10
			modifier = {
				trigger = { has_variable = gw_dissatisfied_with_order }
				add = 5
			}
		}
	}

	option = {
		name = gw_crisis.1100.b
		# Even if leader defers, the atmospheric chain continues
		trigger_event = { id = gw_crisis.1101 months = 6 }
		ai_chance = {
			base = 4
			modifier = {
				trigger = { has_variable = gw_satisfied_with_order }
				add = 6
			}
		}
	}
}

# PHASE 2: DIPLOMATIC INCIDENT AND ULTIMATUM (Tension >= 70/75%)
# ============================================================

# Bloc leaders decide whether to escalate the diplomatic incident.
gw_crisis.2001 = {
	type = country_event

	title = gw_crisis.2001.t
	desc = gw_crisis.2001.d
	flavor = gw_crisis.2001.f

	event_image = {
		video = "gfx/event_pictures/africa_diplomats_negotiating.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	trigger = {
		country_rank >= rank_value:major_power
		gw_is_great_war_bloc_leader = yes
		NOT = { has_variable = gw_ultimatum_seen }
		global_var:gw_world_tension >= 70
		NOT = { has_global_variable = gw_great_war_active }
	}

	immediate = {
		set_variable = gw_ultimatum_seen
	}

	option = {
		name = gw_crisis.2001.a
		default_option = yes
		custom_tooltip = { text = gw_crisis_2001_a_tt }
		add_modifier = {
			name = gw_diplomatic_crisis
			months = 12
		}
		if = {
			limit = { has_modifier = gw_central_powers_leader }
			set_global_variable = gw_ultimatum_issued_central
		}
		if = {
			limit = { has_modifier = gw_allied_powers_leader }
			set_global_variable = gw_ultimatum_issued_allied
		}
		ai_chance = {
			base = 6
			modifier = {
				trigger = { has_modifier = gw_central_powers_leader }
				add = 10
			}
			modifier = {
				trigger = { global_var:gw_world_tension >= 85 }
				add = 8
			}
		}
	}

	option = {
		name = gw_crisis.2001.b
		custom_tooltip = { text = gw_crisis_2001_b_tt }
		add_modifier = {
			name = gw_backchannel_diplomacy
			months = 12
		}
		change_global_variable = { name = gw_world_tension add = -2 }
		ai_chance = {
			base = 8
			modifier = {
				trigger = { has_modifier = gw_allied_powers_leader }
				add = 8
			}
			modifier = {
				trigger = { global_var:gw_world_tension < 85 }
				add = 6
			}
		}
	}

	option = {
		name = gw_crisis.2001.c
		custom_tooltip = { text = gw_crisis_2001_c_tt }
		add_modifier = {
			name = gw_partial_mobilization_plan
			months = 18
		}
		add_modifier = {
			name = gw_diplomatic_crisis
			months = 12
		}
		if = {
			limit = { has_modifier = gw_central_powers_leader }
			set_global_variable = gw_ultimatum_issued_central
		}
		if = {
			limit = { has_modifier = gw_allied_powers_leader }
			set_global_variable = gw_ultimatum_issued_allied
		}
		ai_chance = {
			base = 5
			modifier = {
				trigger = { has_modifier = gw_central_powers_leader }
				add = 8
			}
			modifier = {
				trigger = { global_var:gw_world_tension >= 90 }
				add = 8
			}
		}
	}
}
