principle_deregulation_1 = {
	icon = "gfx/interface/icons/principles_icons/deregulation.dds"
	background = "gfx/interface/icons/principles_icons/principle_tier_1.dds"

	visible = {
		has_dlc_feature = power_bloc_features
	}
	member_modifier = {
		interest_group_ig_industrialists_pol_str_mult = 0.50
		country_capitalists_voting_power_add = 50
		country_force_privatization_bool = yes
	}
	
	ai_weight = {
		value = 100
		
		if = {
			limit = {
				ig:ig_industrialists ?= { is_in_government = yes }
			}
			add = 100
		}			
		
		if = {
			limit = {
				root.power_bloc = {
					current_cohesion_number < 40
				}
			}
			subtract = 100
		}

		if = {
			limit = {
				root.power_bloc = {
					current_cohesion_number < 20
				}
			}
			subtract = 200
		}

		if = {
			limit = { has_strategy = ai_strategy_industrial_expansion }
			multiply = 1.5
		}	
		else_if = {
			limit = { has_strategy = ai_strategy_resource_expansion }
			multiply = 1.25
		}	
	}
}
