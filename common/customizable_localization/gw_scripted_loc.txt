GetGWAlignmentStatus = {
	type = country
	random_valid = no

	text = {
		trigger = {
			OR = {
				has_variable = gw_player_central
				has_variable = gw_alignment_central
			}
		}
		localization_key = gw_alignment_status_central
	}
	text = {
		trigger = {
			OR = {
				has_variable = gw_player_allied
				has_variable = gw_alignment_allied
			}
		}
		localization_key = gw_alignment_status_allied
	}
	text = {
		trigger = {
			OR = {
				has_variable = gw_player_neutral
				has_variable = gw_alignment_neutral
			}
		}
		localization_key = gw_alignment_status_neutral
	}
	text = {
		trigger = {
			has_variable = gw_dissatisfied_with_order
			NOT = {
				OR = {
					has_variable = gw_player_allied
					has_variable = gw_alignment_allied
					has_variable = gw_player_neutral
					has_variable = gw_alignment_neutral
				}
			}
		}
		localization_key = gw_alignment_status_central_leaning
	}
	text = {
		trigger = {
			has_variable = gw_satisfied_with_order
			NOT = {
				OR = {
					has_variable = gw_player_central
					has_variable = gw_alignment_central
					has_variable = gw_player_neutral
					has_variable = gw_alignment_neutral
				}
			}
		}
		localization_key = gw_alignment_status_allied_leaning
	}
	text = {
		localization_key = gw_alignment_status_open
	}
}

GetGWAlignmentDominantPull = {
	type = country
	random_valid = no

	text = {
		trigger = {
			gw_hybrid_alignment_central_probability >= gw_hybrid_alignment_allied_probability
			gw_hybrid_alignment_central_probability >= gw_hybrid_alignment_neutral_probability
		}
		localization_key = gw_alignment_dominant_central
	}
	text = {
		trigger = {
			gw_hybrid_alignment_allied_probability > gw_hybrid_alignment_central_probability
			gw_hybrid_alignment_allied_probability >= gw_hybrid_alignment_neutral_probability
		}
		localization_key = gw_alignment_dominant_allied
	}
	text = {
		trigger = {
			gw_hybrid_alignment_neutral_probability > gw_hybrid_alignment_central_probability
			gw_hybrid_alignment_neutral_probability > gw_hybrid_alignment_allied_probability
		}
		localization_key = gw_alignment_dominant_neutral
	}
	text = {
		localization_key = gw_alignment_dominant_none
	}
}
