# --- World Tension & Great Power Dynamics ---

gw_great_power_total = {
	every_country = {
		limit = {
			country_rank >= rank_value:great_power
		}
		add = 1
	}
	min = 1
}

gw_great_power_dissatisfied_count = {
	every_country = {
		limit = {
			country_rank >= rank_value:great_power
			has_variable = gw_dissatisfied_with_order
		}
		add = 1
	}
}

gw_great_power_dissatisfied_percent = {
	value = gw_great_power_dissatisfied_count
	divide = gw_great_power_total
}

gw_order_dominant_count = {
	every_country = {
		limit = {
			country_rank >= rank_value:great_power
			has_modifier = gw_dominating_world_order
		}
		add = 1
	}
}

# Structural dissatisfaction score - country position in the world order.
# Higher = more dissatisfied = Central Powers leaning.
# This value intentionally ignores current bloc alignment so the AI cannot
# create an Allied/Central feedback loop from its own previous stance.
gw_structural_dissatisfaction_score = {
	value = 0
	# Base dissatisfaction if not a dominant power
	if = {
		limit = {
			NOT = { has_modifier = gw_dominating_world_order }
		}
		add = 35
	}
	# Top-ranked non-dominant GP — powerful challenger (ranks 1-3)
	if = {
		limit = {
			NOT = { has_modifier = gw_dominating_world_order }
			global_country_ranking <= 3
		}
		add = 25
	}
	# Mid-ranked non-dominant GP — rising power (ranks 4-6)
	else_if = {
		limit = {
			NOT = { has_modifier = gw_dominating_world_order }
			global_country_ranking <= 6
		}
		add = 20
	}
	# If a dominant power has massive prestige (proxy for hegemonic power)
	if = {
		limit = {
			any_country = {
				global_country_ranking <= 3
				has_modifier = gw_dominating_world_order
				prestige >= 500
			}
		}
		add = 20
	}
	# Military superiority over #1 ranked country.
	# Use power projection here: army_size/navy_size cannot safely compare
	# against another country's scoped value in every UI/script context.
	if = {
		limit = {
			any_country = {
				global_country_ranking = 1
				naval_power_projection < root.naval_power_projection
			}
		}
		add = 15
	}
	if = {
		limit = {
			any_country = {
				global_country_ranking = 1
				army_power_projection < root.army_power_projection
			}
		}
		add = 15
	}
	# Too many dominant powers dilutes the order
	if = {
		limit = {
			root.gw_order_dominant_count > 4
		}
		add = {
			value = root.gw_order_dominant_count
			subtract = 4
			multiply = 10
		}
	}

	min = 0
	max = 100
}

# Current-facing dissatisfaction keeps old consumers working while only adding
# a modest amount of stance inertia. The hybrid model reads the structural
# value directly for its main order calculations.
gw_dissatisfaction_score = {
	value = gw_structural_dissatisfaction_score
	if = {
		limit = {
			is_ai = yes
			has_variable = gw_alignment_central
		}
		add = 10
	}
	if = {
		limit = {
			is_ai = yes
			has_variable = gw_alignment_allied
		}
		add = -10
	}
	if = {
		limit = {
			has_variable = gw_player_central
		}
		add = 25
	}
	if = {
		limit = {
			has_variable = gw_player_allied
		}
		add = -25
	}

	min = 0
	max = 100
}

# Soft historical Central anchors. These are nudges, not rails: diplomacy and
# structural dissatisfaction can still override them.
gw_soft_historical_central_anchor_score = {
	value = 0
	if = {
		limit = {
			OR = {
				AND = { exists = c:PRU this = c:PRU }
				AND = { exists = c:NGF this = c:NGF }
				AND = { exists = c:GER this = c:GER }
			}
		}
		add = 25
	}
	else_if = {
		limit = {
			OR = {
				AND = { exists = c:AUS this = c:AUS }
				AND = { exists = c:SGF this = c:SGF }
			}
		}
		add = 20
	}
	else_if = {
		limit = {
			exists = c:TUR
			this = c:TUR
			gw_is_major_or_greater = yes
			game_date >= 1870.1.1
		}
		add = 15
	}
	else_if = {
		limit = {
			OR = {
				AND = { exists = c:ITA this = c:ITA }
				AND = { exists = c:JAP this = c:JAP }
			}
			gw_structural_dissatisfaction_score >= 55
			NOT = {
				AND = {
					gw_has_allied_leader_scope = yes
					global_var:gw_allied_leader = {
						has_treaty_alliance_with = { TARGET = root }
					}
				}
			}
		}
		add = 10
	}
}

# Count how many GP have military tech ahead of the average
gw_militarized_gp_count = {
	every_country = {
		limit = {
			gw_is_regular_country = yes
			country_rank >= rank_value:great_power
			OR = {
				has_technology_researched = dreadnought_tech
				has_technology_researched = automatic_machine_guns
			}
		}
		add = 1
	}
}
