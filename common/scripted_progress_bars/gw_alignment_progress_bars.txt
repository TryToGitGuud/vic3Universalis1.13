gw_alignment_allied_probability_bar = {
	name = "gw_alignment_allied_probability_bar"
	desc = "gw_alignment_allied_probability_bar_desc"

	default_green = yes

	weekly_progress = {
		if = {
			limit = {
				owner = { has_variable = gw_alignment_allied_bar_current }
			}
			add = {
				desc = "gw_alignment_allied_progress_sync"
				value = owner.gw_hybrid_alignment_allied_probability
				subtract = owner.var:gw_alignment_allied_bar_current
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 100
}
