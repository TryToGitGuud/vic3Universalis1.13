tyrone_morebloc_dynamic_modifier_update = {
	remove_modifier = $modifier_name$
	if = {
		limit = {
			modifier:$custom_modifier_type$ != 0
		}
		add_modifier = {
			name = $modifier_name$
			multiplier = $custom_multiplier$
		}
	}
}
