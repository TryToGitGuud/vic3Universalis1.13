# --- Pre-War Historical Tension: Anglo-German Naval Rivalry ---
gw_hist.1001 = {
	type = country_event

	title = gw_hist.1001.t
	desc = gw_hist.1001.d
	flavor = gw_hist.1001.f

	event_image = {
		video = "gfx/event_pictures/unspecific_steam_ship.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		game_date >= 1880.1.1
		global_var:gw_world_tension >= 60
		NOT = { has_global_variable = gw_great_war_active }
		NOT = { has_global_variable = gw_call_to_war_active }
		NOT = { has_variable = gw_hist_naval_rivalry_seen }
	}

	immediate = {
		set_variable = gw_hist_naval_rivalry_seen
	}

	option = {
		name = gw_hist.1001.a
		custom_tooltip = { text = gw_hist_1001_a_tt }
		add_modifier = { name = gw_naval_estimates months = 12 }
		gw_hist_tension_add_1 = yes
		ai_chance = {
			base = 8
			modifier = { trigger = { has_variable = gw_dissatisfied_with_order } add = 6 }
			modifier = { trigger = { global_var:gw_world_tension >= 80 } add = 4 }
		}
	}

	option = {
		name = gw_hist.1001.b
		default_option = yes
		custom_tooltip = { text = gw_hist_1001_b_tt }
		add_modifier = { name = gw_last_conference_spirit months = 9 }
		gw_hist_tension_remove_1 = yes
		ai_chance = {
			base = 7
			modifier = { trigger = { has_variable = gw_satisfied_with_order } add = 6 }
		}
	}
}
