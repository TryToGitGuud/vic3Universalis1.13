# GREAT WAR MOD - AI Wartime Decisions
# Purpose: AI-only fallback actions. Players use journal buttons.
# ============================================================

gw_war_propaganda_decision = {
	is_shown = {
		is_ai = yes
		OR = {
			is_country_type = recognized
			is_country_type = colonial
		}
		has_global_variable = gw_great_war_active
		OR = {
			has_variable = gw_war_side_1
			has_variable = gw_war_side_2
		}
	}
	possible = {
		is_at_war = yes
		NOT = { has_modifier = gw_war_propaganda }
	}
	when_taken = {
		gw_apply_war_propaganda = yes
	}
	ai_chance = {
		value = gw_ai_propaganda_desire
	}
}
